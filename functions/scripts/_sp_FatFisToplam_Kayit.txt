CREATE proc [dbo].[sp_FatFisToplam_Kayit]
(
    @FatFisRefNo int
)
AS
INSERT INTO FATFISTOPLAM
(
FFTTAR,
FFTREFNO,
FFTTIPI,
FFTGCFLAG,
FFTKAYONC,
FFTKAYNAK,
FFTCARKOD,
FFTEVRAKNO1,
FFTKONU,
FFTBASLIK,
FFTMATRAH,
FFTTUTAR,
FFTKDVORAN
)
select
FF.FATFISTAR as FFTTAR,
FF.FATFISREFNO as FFTREFNO,
FF.FATFISTIPI as FFTTIPI,
FF.FATFISGCFLAG as FFTGCFLAG,
FF.FATFISKAYONC as FFTKAYONC,
FF.FATFISKAYNAK as FFTKAYNAK,
FF.FATFISCARKOD as FFTCARKOD,
FF.FATFISEVRAKNO1 as FFTEVRAKNO1,
0 as FFTKONU,
'KDV(%' + CAST(CAST(FH.FATHARKDVYUZ AS decimal) AS VARCHAR) +')' AS FFTBASLIK,
FH.ToplamMatrah AS FFTMATRAH,
FH.ToplamKdv as FFTTUTAR,
FH.FATHARKDVYUZ as FFTKDVORAN
from FATFIS FF inner join
(
select FATHARREFNO, FATHARKDVYUZ, sum(FATHARKDVMATRAH) as ToplamMatrah, SUM(FATHARKDVTUTAR) as ToplamKdv
from FATHAR where FATHARREFNO = @FatFisRefNo group by FATHARREFNO, FATHARKDVYUZ
) as FH on FF.FATFISREFNO = FH.FATHARREFNO
update FATFISTOPLAM set FFTKONU = 101
WHERE FFTREFNO = @FatFisRefNo and FFTKONU = 0 and FFTKDVORAN = (SELECT TOP 1 FFTKDVORAN FROM FATFISTOPLAM where FFTREFNO = @FatFisRefNo and FFTKONU = 0 ORDER BY FFTKDVORAN asc)
while exists(select * from FATFISTOPLAM where FFTREFNO = @FatFisRefNo and FFTKONU = 0)
begin
declare @FFTKONU int 
set @FFTKONU = (select max(FFTKONU) + 1 from FATFISTOPLAM where FFTREFNO = @FatFisRefNo)
update FATFISTOPLAM set FFTKONU = @FFTKONU
WHERE FFTREFNO = @FatFisRefNo and FFTKONU = 0 and FFTKDVORAN = (SELECT TOP 1 FFTKDVORAN FROM FATFISTOPLAM where FFTREFNO = @FatFisRefNo and FFTKONU = 0 ORDER BY FFTKDVORAN asc)
end
